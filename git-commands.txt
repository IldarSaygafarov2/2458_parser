git -v - проверяет версию гита на компе

.gitignore - описываем какие папки и файлы не отправлять на гитхаб

git init - инициализация пустого репозитория в проекте
git status - проверка состояния файлов внутри проекта
git add . - Добавление файлов для просмотра гитом
git config user.name "username" - указываем username аккаунта который отправляет изменения
git config user.email "email" - указываем email аккаунта который отправляет изменения

git config --global user.name "username" - глобально указываем username аккаунта который отправляет изменения
git config --global user.email "email" - глобольно указываем email аккаунта который отправляет изменения

git commit -m "message" - комментарий к отправляемым файлам

(выполняется 1 раз для 1 проекта)
git branch -M main - создание главной ветки
git remote add origin ссылка_с_репозитория - связывание локального репозитория с репозиторием на гитхабе
git push -u origin main - отправка изменений в репозиторий
